# 03 条件变量(Condition Variable)

条件变量用于"**等待某个条件成立**",必须和一个互斥锁配合使用:锁保护条件(共享状态),条件变量负责睡眠与唤醒。

## 函数一览

| 函数 | 作用 |
|---|---|
| `PTHREAD_COND_INITIALIZER` | 静态初始化 |
| `pthread_cond_init` / `pthread_cond_destroy` | 动态初始化/销毁 |
| `pthread_cond_wait` | 原子地"释放锁并睡眠",被唤醒后重新加锁再返回 |
| `pthread_cond_timedwait` | 带绝对时间超时的等待,超时返回 `ETIMEDOUT` |
| `pthread_cond_signal` | 唤醒至少一个等待线程 |
| `pthread_cond_broadcast` | 唤醒所有等待线程 |
| `pthread_condattr_init/destroy` | 条件变量属性 |
| `pthread_condattr_setclock` | 设置 timedwait 使用的时钟(如 `CLOCK_MONOTONIC`) |
| `pthread_condattr_setpshared` | 跨进程共享 |

## 标准模板(务必记住)

```c
// 等待方
pthread_mutex_lock(&mu);
while (!condition)                       // 必须用 while,不能用 if
    pthread_cond_wait(&cv, &mu);
/* 条件成立,持有锁,可安全使用共享数据 */
pthread_mutex_unlock(&mu);

// 通知方
pthread_mutex_lock(&mu);
condition = true;                        // 先修改条件
pthread_mutex_unlock(&mu);
pthread_cond_signal(&cv);                // 解锁后再通知也可以(且往往更高效)
```

为什么用 `while`:

1. **虚假唤醒**:`pthread_cond_wait` 可能在没人 signal 的情况下返回。
2. **条件被抢占**:被唤醒到重新拿到锁之间,别的线程可能已经把条件改回去了。
3. `broadcast` 唤醒多个线程,但资源可能只够一个线程用。

## 1. 最简示例:等待一个标志

```c
#include <stdio.h>
#include <unistd.h>
#include <pthread.h>

static pthread_mutex_t mu = PTHREAD_MUTEX_INITIALIZER;
static pthread_cond_t  cv = PTHREAD_COND_INITIALIZER;
static int ready = 0;

static void *waiter(void *arg) {
    (void)arg;
    pthread_mutex_lock(&mu);
    while (!ready)
        pthread_cond_wait(&cv, &mu);
    printf("waiter: go!\n");
    pthread_mutex_unlock(&mu);
    return NULL;
}

int main(void) {
    pthread_t t;
    pthread_create(&t, NULL, waiter, NULL);

    sleep(1);
    pthread_mutex_lock(&mu);
    ready = 1;
    pthread_mutex_unlock(&mu);
    pthread_cond_signal(&cv);

    pthread_join(t, NULL);
    return 0;
}
```

## 2. 经典:有界缓冲区(生产者-消费者)

两个条件变量:`not_full`(生产者等)、`not_empty`(消费者等)。

```c
#include <stdio.h>
#include <pthread.h>

#define CAP   4
#define ITEMS 12

typedef struct {
    int buf[CAP];
    int head, tail, count;
    pthread_mutex_t mu;
    pthread_cond_t  not_full;
    pthread_cond_t  not_empty;
} queue_t;

static queue_t q = {
    .mu = PTHREAD_MUTEX_INITIALIZER,
    .not_full = PTHREAD_COND_INITIALIZER,
    .not_empty = PTHREAD_COND_INITIALIZER,
};

static void put(int v) {
    pthread_mutex_lock(&q.mu);
    while (q.count == CAP)
        pthread_cond_wait(&q.not_full, &q.mu);
    q.buf[q.tail] = v;
    q.tail = (q.tail + 1) % CAP;
    q.count++;
    pthread_cond_signal(&q.not_empty);
    pthread_mutex_unlock(&q.mu);
}

static int get(void) {
    pthread_mutex_lock(&q.mu);
    while (q.count == 0)
        pthread_cond_wait(&q.not_empty, &q.mu);
    int v = q.buf[q.head];
    q.head = (q.head + 1) % CAP;
    q.count--;
    pthread_cond_signal(&q.not_full);
    pthread_mutex_unlock(&q.mu);
    return v;
}

static void *producer(void *arg) {
    (void)arg;
    for (int i = 0; i < ITEMS; i++) {
        put(i);
        printf("produced %d\n", i);
    }
    return NULL;
}

static void *consumer(void *arg) {
    (void)arg;
    for (int i = 0; i < ITEMS; i++)
        printf("    consumed %d\n", get());
    return NULL;
}

int main(void) {
    pthread_t p, c;
    pthread_create(&p, NULL, producer, NULL);
    pthread_create(&c, NULL, consumer, NULL);
    pthread_join(p, NULL);
    pthread_join(c, NULL);
    return 0;
}
```

## 3. timedwait:超时等待

`timedwait` 默认用 `CLOCK_REALTIME` 的**绝对时间**。系统时间被调整会影响超时,更稳妥的做法是改用 `CLOCK_MONOTONIC`。

```c
#include <stdio.h>
#include <errno.h>
#include <time.h>
#include <pthread.h>

static pthread_mutex_t mu = PTHREAD_MUTEX_INITIALIZER;
static pthread_cond_t  cv;                    // 动态初始化以便设置时钟
static int ready = 0;

int main(void) {
    pthread_condattr_t attr;
    pthread_condattr_init(&attr);
    pthread_condattr_setclock(&attr, CLOCK_MONOTONIC);   // 不受系统改时间影响
    pthread_cond_init(&cv, &attr);
    pthread_condattr_destroy(&attr);

    struct timespec deadline;
    clock_gettime(CLOCK_MONOTONIC, &deadline);
    deadline.tv_sec += 2;                                // 2 秒后超时

    pthread_mutex_lock(&mu);
    int rc = 0;
    while (!ready && rc != ETIMEDOUT)                    // 循环中检查条件和超时
        rc = pthread_cond_timedwait(&cv, &mu, &deadline);

    if (rc == ETIMEDOUT) printf("timeout, condition not met\n");
    else                 printf("condition met\n");
    pthread_mutex_unlock(&mu);

    pthread_cond_destroy(&cv);
    return 0;
}
```

## 4. broadcast:唤醒所有等待者(发令枪)

```c
#include <stdio.h>
#include <unistd.h>
#include <pthread.h>

static pthread_mutex_t mu = PTHREAD_MUTEX_INITIALIZER;
static pthread_cond_t  cv = PTHREAD_COND_INITIALIZER;
static int go = 0;

static void *runner(void *arg) {
    long id = (long)arg;
    pthread_mutex_lock(&mu);
    while (!go)
        pthread_cond_wait(&cv, &mu);
    pthread_mutex_unlock(&mu);
    printf("runner %ld starts\n", id);
    return NULL;
}

int main(void) {
    pthread_t t[4];
    for (long i = 0; i < 4; i++) pthread_create(&t[i], NULL, runner, (void *)i);

    sleep(1);
    pthread_mutex_lock(&mu);
    go = 1;
    pthread_cond_broadcast(&cv);             // 一次唤醒所有等待线程
    pthread_mutex_unlock(&mu);

    for (int i = 0; i < 4; i++) pthread_join(t[i], NULL);
    return 0;
}
```

`signal` 与 `broadcast` 的选择:

- 所有等待者等的是**同一种条件**,且一次只能一个线程处理:`signal`。
- 多个线程等**不同条件**共用一个条件变量,或状态变化可能让多个线程同时满足:`broadcast`。
- 拿不准时用 `broadcast`,正确性优先(`while` 循环会让不满足的线程继续睡)。

## 5. 常见错误

| 错误 | 后果 | 正确做法 |
|---|---|---|
| 用 `if` 而不是 `while` 检查条件 | 虚假唤醒时逻辑错误 | 永远 `while` |
| 修改条件时不持锁 | 丢失唤醒(lost wakeup):等待者检查完条件、尚未睡眠时通知已发出 | 修改条件必须在锁内 |
| `wait` 时没持有互斥锁 | 未定义行为 | 先 `lock` 再 `wait` |
| 不同线程用不同的锁配同一个条件变量 | 未定义行为 | 同一个条件变量只配同一把锁 |
| 把条件变量当作"计数信号量" | signal 发在 wait 之前就丢失了 | 条件状态要存在变量里,条件变量只负责唤醒 |
| 销毁时仍有线程在等待 | 未定义行为 | 先通知并 join 再销毁 |

## 小结

- 条件变量本身不存储状态,**状态存在共享变量中,由互斥锁保护**。
- 模板:`lock → while(!cond) wait → 使用数据 → unlock`。
- 需要超时用 `timedwait`,并优先考虑 `CLOCK_MONOTONIC`。
