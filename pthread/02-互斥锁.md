# 02 互斥锁(Mutex)

互斥锁保证同一时刻只有一个线程进入临界区,用于保护共享数据。

## 函数一览

| 函数 | 作用 |
|---|---|
| `PTHREAD_MUTEX_INITIALIZER` | 静态初始化宏(全局/static 变量) |
| `pthread_mutex_init` | 动态初始化,可带属性 |
| `pthread_mutex_destroy` | 销毁(必须在未加锁状态) |
| `pthread_mutex_lock` | 加锁,拿不到则阻塞 |
| `pthread_mutex_trylock` | 尝试加锁,拿不到立即返回 `EBUSY` |
| `pthread_mutex_timedlock` | 带绝对时间超时的加锁,超时返回 `ETIMEDOUT` |
| `pthread_mutex_unlock` | 解锁 |
| `pthread_mutexattr_init/destroy` | 属性对象的初始化/销毁 |
| `pthread_mutexattr_settype/gettype` | 设置锁类型 |
| `pthread_mutexattr_setpshared` | 是否跨进程共享 |
| `pthread_mutexattr_setrobust` | 持锁线程死亡后的健壮性 |
| `pthread_mutex_consistent` | 将 robust 锁标记为一致 |
| `pthread_mutex_getprioceiling/setprioceiling` | 优先级天花板(较少用) |

## 1. 基本用法:解决数据竞争

### 没有锁:结果不确定

```c
// counter++ 实际是 读 -> 加 -> 写 三步,多线程交错会丢失更新
```

### 加锁版本

```c
#include <stdio.h>
#include <pthread.h>

#define THREADS 4
#define LOOPS   100000

static long counter = 0;
static pthread_mutex_t lock = PTHREAD_MUTEX_INITIALIZER;   // 静态初始化

static void *inc(void *arg) {
    (void)arg;
    for (int i = 0; i < LOOPS; i++) {
        pthread_mutex_lock(&lock);
        counter++;                                          // 临界区
        pthread_mutex_unlock(&lock);
    }
    return NULL;
}

int main(void) {
    pthread_t tid[THREADS];
    for (int i = 0; i < THREADS; i++) pthread_create(&tid[i], NULL, inc, NULL);
    for (int i = 0; i < THREADS; i++) pthread_join(tid[i], NULL);
    printf("counter = %ld (expect %d)\n", counter, THREADS * LOOPS);
    return 0;
}
```

> 仅对一个简单计数器,`<stdatomic.h>` 的 `atomic_fetch_add` 更轻量;锁适合保护多个变量组成的不变量。

## 2. 动态初始化与销毁

用于锁嵌在堆上结构体中的情况:

```c
#include <stdio.h>
#include <stdlib.h>
#include <pthread.h>

typedef struct {
    pthread_mutex_t mu;
    int balance;
} account_t;

account_t *account_new(int init) {
    account_t *a = malloc(sizeof *a);
    pthread_mutex_init(&a->mu, NULL);
    a->balance = init;
    return a;
}

void account_free(account_t *a) {
    pthread_mutex_destroy(&a->mu);   // 销毁时不能有线程持有或等待该锁
    free(a);
}

void deposit(account_t *a, int amt) {
    pthread_mutex_lock(&a->mu);
    a->balance += amt;
    pthread_mutex_unlock(&a->mu);
}

int main(void) {
    account_t *a = account_new(100);
    deposit(a, 50);
    printf("balance = %d\n", a->balance);
    account_free(a);
    return 0;
}
```

## 3. trylock:非阻塞尝试

```c
#include <stdio.h>
#include <errno.h>
#include <unistd.h>
#include <pthread.h>

static pthread_mutex_t mu = PTHREAD_MUTEX_INITIALIZER;

static void *holder(void *arg) {
    (void)arg;
    pthread_mutex_lock(&mu);
    sleep(2);                                // 长时间持有
    pthread_mutex_unlock(&mu);
    return NULL;
}

int main(void) {
    pthread_t t;
    pthread_create(&t, NULL, holder, NULL);
    usleep(100 * 1000);                      // 让 holder 先拿到锁

    int rc = pthread_mutex_trylock(&mu);
    if (rc == EBUSY) {
        printf("lock busy, do something else\n");
    } else if (rc == 0) {
        printf("got lock\n");
        pthread_mutex_unlock(&mu);
    }
    pthread_join(t, NULL);
    return 0;
}
```

典型场景:避免死锁(拿不到第二把锁就释放第一把重试)、"拿不到就跳过"的统计/日志。

## 4. timedlock:带超时加锁

超时参数是**绝对时间**(`CLOCK_REALTIME`),不是相对时长。

```c
#include <stdio.h>
#include <errno.h>
#include <time.h>
#include <unistd.h>
#include <pthread.h>

static pthread_mutex_t mu = PTHREAD_MUTEX_INITIALIZER;

static void *holder(void *arg) {
    (void)arg;
    pthread_mutex_lock(&mu);
    sleep(3);
    pthread_mutex_unlock(&mu);
    return NULL;
}

int main(void) {
    pthread_t t;
    pthread_create(&t, NULL, holder, NULL);
    usleep(100 * 1000);

    struct timespec abs;
    clock_gettime(CLOCK_REALTIME, &abs);
    abs.tv_sec += 1;                         // 最多等 1 秒

    int rc = pthread_mutex_timedlock(&mu, &abs);
    if (rc == ETIMEDOUT)
        printf("timeout\n");
    else if (rc == 0) {
        printf("locked\n");
        pthread_mutex_unlock(&mu);
    }
    pthread_join(t, NULL);
    return 0;
}
```

## 5. 互斥锁类型

通过 `pthread_mutexattr_settype` 设置:

| 类型 | 重复加锁 | 非持有者解锁 / 未加锁时解锁 | 特点 |
|---|---|---|---|
| `PTHREAD_MUTEX_NORMAL` | **死锁** | 未定义行为 | 最快,无检查 |
| `PTHREAD_MUTEX_ERRORCHECK` | 返回 `EDEADLK` | 返回 `EPERM` | 调试首选 |
| `PTHREAD_MUTEX_RECURSIVE` | 允许,需解锁相同次数 | 返回 `EPERM` | 可重入 |
| `PTHREAD_MUTEX_DEFAULT` | 未定义(Linux 上等同 NORMAL) | 未定义 | 默认类型 |

### ERRORCHECK 示例:发现自死锁

```c
#include <stdio.h>
#include <string.h>
#include <pthread.h>

int main(void) {
    pthread_mutexattr_t attr;
    pthread_mutex_t mu;

    pthread_mutexattr_init(&attr);
    pthread_mutexattr_settype(&attr, PTHREAD_MUTEX_ERRORCHECK);
    pthread_mutex_init(&mu, &attr);
    pthread_mutexattr_destroy(&attr);

    pthread_mutex_lock(&mu);
    int rc = pthread_mutex_lock(&mu);        // NORMAL 型会永久阻塞,这里直接报错
    printf("second lock: %s\n", strerror(rc));   // Resource deadlock avoided

    pthread_mutex_unlock(&mu);
    pthread_mutex_destroy(&mu);
    return 0;
}
```

### RECURSIVE 示例:递归/互相调用的函数都要加锁

```c
#include <stdio.h>
#include <pthread.h>

static pthread_mutex_t mu;
static int depth_sum = 0;

static void visit(int n) {
    pthread_mutex_lock(&mu);                 // 同一线程可重复加锁
    depth_sum += n;
    if (n > 0) visit(n - 1);
    pthread_mutex_unlock(&mu);
}

int main(void) {
    pthread_mutexattr_t attr;
    pthread_mutexattr_init(&attr);
    pthread_mutexattr_settype(&attr, PTHREAD_MUTEX_RECURSIVE);
    pthread_mutex_init(&mu, &attr);
    pthread_mutexattr_destroy(&attr);

    visit(5);
    printf("sum = %d\n", depth_sum);         // 15
    pthread_mutex_destroy(&mu);
    return 0;
}
```

> 递归锁常被视为设计味道(代码职责不清)。优先把"带锁/不带锁"拆成两个函数,例如 `foo()` 加锁后调用 `foo_locked()`。

## 6. 死锁与规避

### 经典死锁

```
线程 A: lock(m1) -> lock(m2)
线程 B: lock(m2) -> lock(m1)     // 互相等待,永远卡住
```

### 规避策略

1. **固定加锁顺序**:所有线程按相同顺序拿锁(例如按地址或编号排序)。
2. 用 `trylock`,失败就释放已持有的锁并重试。
3. 尽量缩小临界区,不要在持锁时调用不明回调、做阻塞 I/O。
4. 尽量一次只持有一把锁。

### 按地址排序的转账示例

```c
#include <stdio.h>
#include <stdint.h>
#include <pthread.h>

typedef struct { pthread_mutex_t mu; int balance; } acct_t;

static void transfer(acct_t *from, acct_t *to, int amt) {
    acct_t *first  = ((uintptr_t)from < (uintptr_t)to) ? from : to;   // 按地址排序
    acct_t *second = (first == from) ? to : from;

    pthread_mutex_lock(&first->mu);
    pthread_mutex_lock(&second->mu);
    from->balance -= amt;
    to->balance   += amt;
    pthread_mutex_unlock(&second->mu);
    pthread_mutex_unlock(&first->mu);
}

int main(void) {
    acct_t a = { PTHREAD_MUTEX_INITIALIZER, 100 };
    acct_t b = { PTHREAD_MUTEX_INITIALIZER, 100 };
    transfer(&a, &b, 30);
    transfer(&b, &a, 10);
    printf("a=%d b=%d\n", a.balance, b.balance);   // a=80 b=120
    return 0;
}
```

## 7. 进程间共享与健壮锁(进阶)

```c
// 放在共享内存(mmap / shm_open)中的互斥锁
pthread_mutexattr_t attr;
pthread_mutexattr_init(&attr);
pthread_mutexattr_setpshared(&attr, PTHREAD_PROCESS_SHARED);   // 跨进程
pthread_mutexattr_setrobust(&attr, PTHREAD_MUTEX_ROBUST);      // 持锁进程崩溃可恢复
pthread_mutex_init(mutex_in_shm, &attr);

int rc = pthread_mutex_lock(mutex_in_shm);
if (rc == EOWNERDEAD) {
    // 前一个持有者死亡,数据可能不一致:修复数据后调用
    pthread_mutex_consistent(mutex_in_shm);
}
pthread_mutex_unlock(mutex_in_shm);
```

## 小结

- 加锁和解锁必须成对,**所有退出路径**(包括 `return`/`goto`/错误分支)都要解锁。
- 不要对已销毁或未初始化的锁操作;不要拷贝 `pthread_mutex_t`。
- `fork` 后子进程中的锁状态复杂,尽量只在 `fork` 后立即 `exec`(或用 `pthread_atfork` 处理)。
- 临界区越小越好。
